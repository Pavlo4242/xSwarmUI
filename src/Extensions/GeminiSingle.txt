    <script>
    (function() {
        // Debug helper
        const log = (msg) => {
            const d = document.getElementById('p_debug');
            if(d) { d.innerHTML += `<div>${new Date().toLocaleTimeString()} ${msg}</div>`; d.scrollTop = d.scrollHeight; }
            console.log('[Preview] ' + msg);
        };

        // Persistent History
        if (!window.previewHistory) window.previewHistory = [];

        class PreviewTab {
            constructor() {
                log("Init PreviewTab (Watcher Mode)");
                this.isGenerating = false;
                this.debounceTimer = null;
                
                // Map DOM
                this.dom = {
                    prompt: document.getElementById('p_prompt'),
                    neg: document.getElementById('p_neg'),
                    model: document.getElementById('p_model'),
                    steps: document.getElementById('p_steps'),
                    cfg: document.getElementById('p_cfg'),
                    width: document.getElementById('p_width'),
                    height: document.getElementById('p_height'),
                    batch: document.getElementById('p_batch'),
                    seed: document.getElementById('p_seed'),
                    auto: document.getElementById('p_auto'),
                    img: document.getElementById('p_img'),
                    loras: document.getElementById('p_loras'),
                    history: document.getElementById('p_history'),
                    stepsDiv: document.getElementById('p_steps'),
                    
                    // Overlays
                    info: document.getElementById('p_info'),
                    prog: document.getElementById('p_progress'),
                    fill: document.getElementById('p_fill'),
                    err: document.getElementById('p_error')
                };

                this.setupEvents();
                
                // START WATCHER
                setTimeout(() => this.monitorMainTab(), 1000); 
                
                if (window.previewHistory.length > 0) this.refreshHistoryUI();

                // Initial Sync
                setTimeout(() => this.syncFromMain(), 500);
            }

            setupEvents() {
                // Inputs trigger Auto-Gen
                ['prompt', 'neg', 'steps', 'cfg', 'width', 'height', 'batch', 'seed', 'model'].forEach(k => {
                    this.dom[k].addEventListener('input', () => this.trigger());
                });
                this.dom.auto.addEventListener('change', () => this.trigger());

                // Buttons
                document.getElementById('p_gen').onclick = () => { log("Manual Generate"); this.generate(); };
                document.getElementById('p_pull').onclick = () => { log("Pulling..."); this.syncFromMain(); };
                document.getElementById('p_push').onclick = () => { log("Pushing..."); this.syncToMain(); };
                document.getElementById('p_refresh_loras').onclick = () => this.buildLoras();
            }

            // --- THE WATCHER: Captures Main Tab Activity ---
            monitorMainTab() {
                const targetNode = document.getElementById('current_image_batch');
                
                if (!targetNode) {
                    log("Main tab 'current_image_batch' missing. Retrying...");
                    setTimeout(() => this.monitorMainTab(), 2000);
                    return;
                }

                log("Watcher Active on Main Tab");
                
                const observer = new MutationObserver((mutations) => {
                    mutations.forEach((mutation) => {
                        // Check for images
                        const img = targetNode.querySelector('img');
                        if (img && img.src && img.src !== this.dom.img.src) {
                            
                            const src = img.src;
                            this.dom.img.src = src; // Show in main view
                            this.addStepFrame(src); // Add to step strip
                            
                            // If it's a real file (not base64), it's a final image
                            if (!src.startsWith('data:')) {
                                log("Captured Main Tab Image");
                                this.dom.img.classList.remove('generating');
                                this.addToHistory(src, {source: "Main Tab"});
                            } else {
                                this.dom.img.classList.add('generating');
                            }
                        }
                    });
                });

                observer.observe(targetNode, { attributes: true, childList: true, subtree: true });
            }

            trigger() {
                if(!this.dom.auto.checked || this.isGenerating) return;
                clearTimeout(this.debounceTimer);
                this.debounceTimer = setTimeout(() => this.generate(), 500);
            }

            syncFromMain() {
                const getVal = (id) => {
                    const el = document.getElementById(id) || document.getElementById('input_' + id);
                    return el ? el.value : null;
                };
                
                const p = getVal('prompt');
                if (p === null) return; 

                this.dom.prompt.value = p;
                this.dom.neg.value = getVal('negativeprompt') || '';
                this.dom.steps.value = getVal('steps') || 20;
                this.dom.cfg.value = getVal('cfgscale') || 7;
                this.dom.width.value = getVal('width') || 512;
                this.dom.height.value = getVal('height') || 512;
                this.dom.seed.value = getVal('seed') || -1;
                this.dom.batch.value = getVal('batchsize') || 1;

                const mMain = document.getElementById('current_model');
                if(mMain) {
                    this.dom.model.innerHTML = mMain.innerHTML;
                    this.dom.model.value = mMain.value;
                }
                this.buildLoras();
            }

            syncToMain() {
                const setVal = (id, val) => {
                    const el = document.getElementById(id) || document.getElementById('input_' + id);
                    if(el) {
                        el.value = val;
                        el.dispatchEvent(new Event('change', {bubbles:true}));
                        el.dispatchEvent(new Event('input', {bubbles:true}));
                    }
                };
                setVal('prompt', this.dom.prompt.value);
                setVal('negativeprompt', this.dom.neg.value);
                setVal('steps', this.dom.steps.value);
                setVal('cfgscale', this.dom.cfg.value);
                setVal('width', this.dom.width.value);
                setVal('height', this.dom.height.value);
                setVal('seed', this.dom.seed.value);
                setVal('batchsize', this.dom.batch.value);
            }

            buildLoras() {
                this.dom.loras.innerHTML = '';
                if(typeof loraHelper !== 'undefined' && loraHelper.selected) {
                    loraHelper.selected.forEach(l => {
                        const div = document.createElement('div');
                        div.style.background = '#333'; div.style.padding = '5px'; div.style.borderRadius = '4px';
                        div.innerHTML = `
                            <div style="display:flex; justify-content:space-between; font-size:11px; font-weight:bold;">
                                <span style="overflow:hidden; text-overflow:ellipsis; max-width:180px;">${l.name}</span>
                                <span id="plv_${l.name}">${l.weight}</span>
                            </div>
                            <input type="range" min="-2" max="2" step="0.1" value="${l.weight}" style="width:100%"
                                oninput="document.getElementById('plv_${l.name}').innerText=this.value; window.previewTab.trigger()">
                        `;
                        this.dom.loras.appendChild(div);
                    });
                } else {
                    this.dom.loras.innerHTML = '<div style="color:#666; font-size:11px;">No LoRAs Active</div>';
                }
            }

            generate() {
                if(this.isGenerating) return;
                this.isGenerating = true;
                
                // Clear Step Strip
                this.dom.stepsDiv.innerHTML = ''; 
                this.dom.prog.style.display = 'block';
                this.dom.img.classList.add('generating');

                let prompt = this.dom.prompt.value;
                Array.from(this.dom.loras.querySelectorAll('input[type=range]')).forEach(r => {
                     const name = r.previousElementSibling.children[0].innerText;
                     if(parseFloat(r.value) !== 0) prompt += ` <lora:${name}:${r.value}>`;
                });

                const input = {
                    'session_id': localStorage.getItem('session_id'),
                    'images': parseInt(this.dom.batch.value) || 1,
                    'prompt': prompt,
                    'negativeprompt': this.dom.neg.value,
                    'steps': parseInt(this.dom.steps.value),
                    'cfgscale': parseFloat(this.dom.cfg.value),
                    'width': parseInt(this.dom.width.value),
                    'height': parseInt(this.dom.height.value),
                    'seed': parseInt(this.dom.seed.value),
                    'model': this.dom.model.value,
                    'donotsave': true
                };

                makeWSRequestT2I('GenerateText2ImageWS', input, data => {
                    if(data.gen_progress) {
                        const p = data.gen_progress;
                        if(p.preview) {
                            this.dom.img.src = p.preview;
                            this.addStepFrame(p.preview);
                        }
                        if(p.overall_percent) {
                            this.dom.fill.style.width = (p.overall_percent * 100) + '%';
                        }
                    }
                    if(data.image) {
                        const url = data.image.image || data.image;
                        this.dom.img.src = url;
                        this.dom.img.classList.remove('generating');
                        this.addToHistory(url, data.metadata || "{}");
                    }
                    if(data.done || (data.status && data.status.live_gens == 0)) {
                        this.isGenerating = false;
                        this.dom.prog.style.display = 'none';
                        log("Gen Done");
                    }
                });
            }

            addStepFrame(url) {
                // Deduplicate
                if (this.dom.stepsDiv.lastChild && this.dom.stepsDiv.lastChild.querySelector('img').src === url) return;

                const div = document.createElement('div');
                div.className = 'p-step-item';
                div.innerHTML = `<img src="${url}">`;
                div.onclick = () => {
                    this.dom.img.src = url;
                    Array.from(this.dom.stepsDiv.children).forEach(c => c.classList.remove('active'));
                    div.classList.add('active');
                };
                this.dom.stepsDiv.appendChild(div);
                this.dom.stepsDiv.scrollLeft = this.dom.stepsDiv.scrollWidth;
            }

            addToHistory(url, metadata) {
                window.previewHistory.push({ url, metadata, timestamp: Date.now() });
                this.refreshHistoryUI();
            }

            refreshHistoryUI() {
                this.dom.history.innerHTML = '';
                for (let i = window.previewHistory.length - 1; i >= 0; i--) {
                    const item = window.previewHistory[i];
                    
                    const div = document.createElement('div');
                    div.className = 'p-history-item';
                    
                    const img = document.createElement('img');
                    img.src = item.url;
                    div.appendChild(img);

                    const overlay = document.createElement('div');
                    overlay.className = 'p-hist-overlay';
                    
                    const reuseBtn = document.createElement('i');
                    reuseBtn.className = 'fas fa-redo p-icon-btn';
                    reuseBtn.title = 'Reuse Params';
                    reuseBtn.onclick = (e) => {
                        e.stopPropagation();
                        this.restoreParams(item.metadata);
                    };
                    overlay.appendChild(reuseBtn);

                    div.onclick = () => {
                        this.dom.img.src = item.url;
                        Array.from(this.dom.history.children).forEach(c => c.classList.remove('active'));
                        div.classList.add('active');
                    };

                    div.appendChild(overlay);
                    this.dom.history.appendChild(div);
                }
            }

            restoreParams(metadataStr) {
                try {
                    const meta = typeof metadataStr === 'string' ? JSON.parse(metadataStr) : metadataStr;
                    log("Restoring Params...");
                    if(meta.prompt) this.dom.prompt.value = meta.prompt;
                    if(meta.negativeprompt) this.dom.neg.value = meta.negativeprompt;
                    if(meta.steps) this.dom.steps.value = meta.steps;
                    if(meta.cfgscale) this.dom.cfg.value = meta.cfgscale;
                    if(meta.width) this.dom.width.value = meta.width;
                    if(meta.height) this.dom.height.value = meta.height;
                    if(meta.seed) this.dom.seed.value = meta.seed;
                    if(meta.model) this.dom.model.value = meta.model;
                } catch(e) {
                    log("Error restoring params");
                }
            }
        }

        window.previewTab = new PreviewTab();
    })();
    </script>
</div>
